    17→/* ─── Animated counter ─── */
    18→function Counter({ target, suffix = '' }: { target: number; suffix?: string }) {
    19→  const [count, setCount] = useState(0);
    20→  useEffect(() => {
    21→    let start = 0;
    22→    const duration = 2000;
    23→    const step = Math.ceil(target / (duration / 16));
    24→    const timer = setInterval(() => {
    25→      start += step;
    26→      if (start >= target) { start = target; clearInterval(timer); }
    27→      setCount(start);
    28→    }, 16);
    29→    return () => clearInterval(timer);
    30→  }, [target]);
    31→  return <span>{count.toLocaleString()}{suffix}</span>;
    32→}
