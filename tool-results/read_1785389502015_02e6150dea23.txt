    17→    17→/* ─── Animated counter (fixed: uses ceil on step, handles 99→99.9 correctly) ─── */
    18→    18→function Counter({ target, suffix = '', decimals = 0 }: { target: number; suffix?: string; decimals?: number }) {
    19→    19→  const [count, setCount] = useState(0);
    20→    20→  useEffect(() => {
    21→    21→    let start = 0;
    22→    22→    const duration = 2000;
    23→    23→    const totalSteps = Math.ceil(duration / 16);
    24→    24→    const rawStep = (target * Math.pow(10, decimals)) / totalSteps;
    25→    25→    const step = Math.max(rawStep, 0.01);
    26→    26→    const timer = setInterval(() => {
    27→    27→      start += step;
    28→    28→      if (start >= target * Math.pow(10, decimals)) {
    29→    29→        start = target * Math.pow(10, decimals);
    30→    30→        clearInterval(timer);
    31→    31→      }
    32→    32→      setCount(start);
    33→    33→    }, 16);
    34→    34→    return () => clearInterval(timer);
    35→    35→  }, [target, decimals]);
    36→    36→  return <span>{(count / Math.pow(10, decimals)).toFixed(decimals)}{suffix}</span>;
    37→    37→}
